func.func @package_export_full_instr(
    %input: memref<8xf32, #wafer.memory<ddr, tensor>>,
    %output: memref<8xf32, #wafer.memory<ddr, tensor>>) {
  %zero = arith.constant 0.0 : f32
  %src = memref.alloc() {wafer.spm.offset = #wafer.spm_offset<65536>}
      : memref<8xf32, #wafer.memory<spm, tensor>>
  %tmp = memref.alloc() {wafer.spm.offset = #wafer.spm_offset<65792>}
      : memref<8xf32, #wafer.memory<spm, tensor>>
  %red_src = memref.alloc() {wafer.spm.offset = #wafer.spm_offset<66048>}
      : memref<1x8xf32, #wafer.memory<spm, cx>>
  %red = memref.alloc() {wafer.spm.offset = #wafer.spm_offset<66304>}
      : memref<1xf32, #wafer.memory<spm, cx>>
  %converted = memref.alloc() {wafer.spm.offset = #wafer.spm_offset<66560>}
      : memref<8xi32, #wafer.memory<spm, tensor>>
  %pred = memref.alloc() {wafer.spm.offset = #wafer.spm_offset<66816>}
      : memref<8xi1, #wafer.memory<spm, tensor>>
  %mask = memref.alloc() {wafer.spm.offset = #wafer.spm_offset<67072>}
      : memref<8xf32, #wafer.memory<spm, tensor>>
  %act = memref.alloc() {wafer.spm.offset = #wafer.spm_offset<67328>}
      : memref<1x8x8x64xf32, #wafer.memory<spm, ncx>>
  %weight = memref.alloc() {wafer.spm.offset = #wafer.spm_offset<83712>}
      : memref<3x3x64x64xf32, #wafer.memory<spm, ncx>>
  %conv_out = memref.alloc() {wafer.spm.offset = #wafer.spm_offset<231168>}
      : memref<1x8x8x64xf32, #wafer.memory<spm, ncx>>
  %pool_out = memref.alloc() {wafer.spm.offset = #wafer.spm_offset<247552>}
      : memref<1x4x4x64xf32, #wafer.memory<spm, ncx>>
  %pool_idx = memref.alloc() {wafer.spm.offset = #wafer.spm_offset<251648>}
      : memref<1x4x4x64xi32, #wafer.memory<spm, ncx>>
  %moved = memref.alloc() {wafer.spm.offset = #wafer.spm_offset<255744>}
      : memref<1x8x8x64xf32, #wafer.memory<spm, tensor>>
  %arg_value = memref.alloc() {wafer.spm.offset = #wafer.spm_offset<272128>}
      : memref<1xf32, #wafer.memory<spm, tensor>>
  %arg_index = memref.alloc() {wafer.spm.offset = #wafer.spm_offset<272384>}
      : memref<1xi32, #wafer.memory<spm, tensor>>
  wafer.instr.fill %src, %zero
      : memref<8xf32, #wafer.memory<spm, tensor>>, f32
  wafer.instr.elementwise #wafer.instr_elementwise_kind<eq> %src, %tmp into %tmp
      : memref<8xf32, #wafer.memory<spm, tensor>>,
        memref<8xf32, #wafer.memory<spm, tensor>>
      into memref<8xf32, #wafer.memory<spm, tensor>>
  wafer.instr.bit2fp %pred into %mask
      : memref<8xi1, #wafer.memory<spm, tensor>>
      to memref<8xf32, #wafer.memory<spm, tensor>>
  wafer.instr.mask_move %src, %mask into %tmp
      : memref<8xf32, #wafer.memory<spm, tensor>>,
        memref<8xf32, #wafer.memory<spm, tensor>>
      into memref<8xf32, #wafer.memory<spm, tensor>>
  wafer.instr.reduce #wafer.instr_reduce_kind<avg> %red_src into %red, %zero : f32
      {dim = 0 : i64}
      : memref<1x8xf32, #wafer.memory<spm, cx>>
      into memref<1xf32, #wafer.memory<spm, cx>>
  wafer.instr.reduce #wafer.instr_reduce_kind<max> %red_src into %red
      {dim = 0 : i64, init_value = 0xFF800000 : f32}
      : memref<1x8xf32, #wafer.memory<spm, cx>>
      into memref<1xf32, #wafer.memory<spm, cx>>
  wafer.instr.convert #wafer.instr_convert_kind<fp32_int32> %tmp into %converted
      {rounding_mode = 0 : i64}
      : memref<8xf32, #wafer.memory<spm, tensor>>
      to memref<8xi32, #wafer.memory<spm, tensor>>
  wafer.instr.conv #wafer.instr_conv_kind<conv> %act, %weight into %conv_out
      {input_shape = array<i64: 1, 8, 8, 64>,
       weight_shape = array<i64: 3, 3, 64, 64>,
       output_shape = array<i64: 1, 8, 8, 64>,
       pads = array<i64: 1, 1, 1, 1>,
       unpads = array<i64: 0, 0, 0, 0>,
       kernel_strides = array<i64: 3, 3, 1, 1>,
       dilations = array<i64: 1, 1>}
      : memref<1x8x8x64xf32, #wafer.memory<spm, ncx>>,
        memref<3x3x64x64xf32, #wafer.memory<spm, ncx>>
      into memref<1x8x8x64xf32, #wafer.memory<spm, ncx>>
  wafer.instr.pool #wafer.instr_pool_kind<indexedmax> %act into %pool_out, %pool_idx
      {source_shape = array<i64: 1, 8, 8, 64>,
       dest_shape = array<i64: 1, 4, 4, 64>,
       pads = array<i64: 0, 0, 0, 0>,
       kernel_strides = array<i64: 2, 2, 2, 2>}
      : memref<1x8x8x64xf32, #wafer.memory<spm, ncx>>
      into memref<1x4x4x64xf32, #wafer.memory<spm, ncx>>,
           memref<1x4x4x64xi32, #wafer.memory<spm, ncx>>
  wafer.instr.unpool #wafer.instr_unpool_kind<mask> %pool_out into %act
      {source_shape = array<i64: 1, 4, 4, 64>,
       dest_shape = array<i64: 1, 8, 8, 64>,
       kernel_strides = array<i64: 2, 2, 2, 2>,
       index = 0 : i64}
      : memref<1x4x4x64xf32, #wafer.memory<spm, ncx>>
      into memref<1x8x8x64xf32, #wafer.memory<spm, ncx>>
  wafer.instr.tdma_data_move #wafer.instr_data_move_kind<pad> %act into %moved
      {source_shape = array<i64: 1, 8, 8, 64>,
       dest_shape = array<i64: 1, 8, 8, 64>,
       pads = array<i64: 0, 0, 0, 0>}
      : memref<1x8x8x64xf32, #wafer.memory<spm, ncx>>
      to memref<1x8x8x64xf32, #wafer.memory<spm, tensor>>
  wafer.instr.peripheral #wafer.instr_peripheral_kind<argmax> %tmp into %arg_value, %arg_index
      {elem_count = 8 : i64}
      : memref<8xf32, #wafer.memory<spm, tensor>>
      into memref<1xf32, #wafer.memory<spm, tensor>>,
           memref<1xi32, #wafer.memory<spm, tensor>>
  %send = wafer.instr.dte_send %src {peer = 1 : i64, bytes = 32 : i64}
      : memref<8xf32, #wafer.memory<spm, tensor>> -> !async.token
  %recv = wafer.instr.dte_recv %tmp {peer = 0 : i64, bytes = 32 : i64}
      : memref<8xf32, #wafer.memory<spm, tensor>> -> !async.token
  wafer.instr.dte_wait %send, %recv : !async.token, !async.token
  wafer.instr.local_fence
  return
}
